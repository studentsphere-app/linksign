  type={{

## Usage

```typescript
import { getCourseByCode } from "@studentsphere/linksign";

async function run() {
  const result = await getCourseByCode('YOUR_TOKEN', 'AUTH_CODE', 'OPTIONAL_DEVICE_ID');
  console.log(result);
}

run().catch(console.error);
```

## Return Value

Returns a `Promise` resolving to a **[CourseEntry](/docs/types/planning#courseentry-interface)** object.
