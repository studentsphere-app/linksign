## Usage

```typescript
import { getPublicContactsDirectory } from "@studentsphere/linksign";

async function run() {
  const result = await getPublicContactsDirectory('SCHOOL_ID', { name: 'John Doe', page: 1 });

}

run().catch(console.error);
```
